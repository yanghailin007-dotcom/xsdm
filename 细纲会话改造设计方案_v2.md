# 细纲会话改造设计方案 v2.0

## 一、现状分析

### 1.1 当前系统架构
```
市场导向创作流程：
题材分析 → 模式选择 → 对话打磨(7步) → 生成最终方案 → 第一阶段产物生成
                                                    ↓
                                            战术规划(TacticalPlanner)
                                                    ↓
                                            章节生成(每30章滚动规划)
```

### 1.2 存在的问题
| 问题 | 现状 | 影响 |
|------|------|------|
| 角色管理分散 | 角色设计在 phase_one_products/角色设计.json | 战术规划时无法有效利用已有角色 |
| 战术规划单一 | 仅基于情绪曲线和阶段目标 | 缺乏角色出场规划，导致章节生成时随意创造新角色 |
| 核心设定丢失 | 金手指、卖点等未在战术层强调 | 章节生成偏离原作设定 |

### 1.3 角色数据现状
**存储位置**：`phase_one_products/角色设计.json`
```json
{
  "protagonist": { "name": "沈浪", "traits": [...] },
  "core_allies": [ { "name": "二哈", "role": "..." }, ... ],
  "main_antagonists": { "early_stage": [...], "mid_stage": [...] }
}
```

## 二、设计方案

### 2.1 总体架构

```
细纲会话系统 (TacticalDialogSession)
├── 第1轮：情节规划 (复用现有TacticalPlanner)
│   └── 输出：30章情节大纲 + 情绪曲线
├── 第2轮：角色规划 (新增)
│   ├── 输入：情节大纲 + 已有角色列表
│   ├── 处理：AI对话选择角色出场
│   └── 输出：角色出场规划表
└── 合并输出：完整战术蓝图
    ├── chapters: 30章详细信息
    └── characters: { core, major, minor, new }
```

# web/services/market_driven/character_loader.py

class CharacterLoader:
    """从phase_one_products加载角色设计"""
    
    def load_from_project(self, project_path: Path) -> List[Dict]:
        character_file = project_path / "phase_one_products" / "角色设计.json"
        if not character_file.exists():
            return []
        
        with open(character_file, 'r', encoding='utf-8') as f:
            data = json.load(f)
        
        characters = []
        
        # 主角
        if 'protagonist' in data:
            p = data['protagonist']
            characters.append({
                'name': p.get('name'),
                'type': 'protagonist',
                'role': '主角',
                'traits': p.get('traits', []),
                'source': '角色设计.json'
            })
        
        # 盟友
        for ally in data.get('core_allies', []):
            characters.append({
                'name': ally.get('name'),
                'type': 'ally',
                'role': ally.get('role', '配角'),
                'traits': ally.get('traits', []),
                'source': '角色设计.json'
            })
        
        # 敌人
        for stage, villains in data.get('main_antagonists', {}).items():
            for villain in villains:
                characters.append({
                    'name': villain.get('name'),
                    'type': 'villain',
                    'role': f"反派({stage})",
                    'motivation': villain.get('motivation', ''),
                    'source': '角色设计.json'
                })
        
        return characters
```

### 步骤2：创建细纲会话模块

文件：`web/services/market_driven/tactical_dialog_session.py`

```python
"""
细纲会话管理
Tactical Dialog Session v2.0
"""

import json
import logging
import re
from typing import Dict, List, Optional
from datetime import datetime
from pathlib import Path

logger = logging.getLogger(__name__)


class TacticalDialogSession:
    """
    细纲会话 - 两轮对话生成30章规划
    
    流程:
    1. 第1轮：生成30章情节规划
    2. 第2轮：基于情节+已有角色，规划角色出场
    3. 合并输出完整blueprint
    """
    
    def __init__(
        self,
        session_id: str,
        start_chapter: int,
        end_chapter: int,
        novel_data: Dict,
        existing_characters: List[Dict],
        api_client=None,
        project_path: Path = None
    ):
        self.session_id = session_id
        self.start_chapter = start_chapter
        self.end_chapter = end_chapter
        self.novel_data = novel_data
        self.existing_characters = existing_characters
        self.api_client = api_client
        self.project_path = project_path
        
        self.current_round = 0
        self.chapters_plan = None
        self.characters_plan = None
        self.final_blueprint = None
    
    def generate_full_blueprint(self) -> Dict:
        """完成会话的两轮对话"""
        logger.info(f"[细纲会话 {self.session_id}] 开始生成")
        
        # 第1轮：情节规划
        self._round_1_plot_planning()
        
        # 第2轮：角色规划
        self._round_2_character_planning()
        
        # 合并
        self._merge_blueprint()
        
        logger.info(f"[细纲会话 {self.session_id}] 生成完成")
        return self.final_blueprint
    
    def _round_1_plot_planning(self):
        """第1轮：生成30章情节规划"""
        from .tactical_planner import TacticalPlanner
        
        planner = TacticalPlanner(self.api_client, self.project_path)
        
        self.chapters_plan = planner.plan_next_batch(
            start_chapter=self.start_chapter,
            end_chapter=self.end_chapter,
            novel_title=self.novel_data.get('title', '未命名'),
            protagonist_name=self._get_protagonist_name(),
            stage_goal=self._get_stage_goal(),
            previous_summary=self._get_previous_summary(),
            emotion_curve=self.novel_data.get('emotion_curve'),
            bestseller_analysis=self.novel_data.get('bestseller_analysis')
        )
        
        self.current_round = 1
    
    def _round_2_character_planning(self):
        """第2轮：基于情节和已有角色，规划角色出场"""
        if not self.api_client:
            self.characters_plan = self._get_default_character_plan()
            return
        
        prompt = self._build_character_dialog_prompt()
        
        try:
            response = self.api_client.generate_content_with_retry(
                content_type="tactical_character_planning",
                user_prompt=prompt,
                system_prompt=self._get_character_system_prompt(),
                temperature=0.7
            )
            
            self.characters_plan = self._parse_character_response(response)
            self.current_round = 2
            
            # 记录新角色
            new_count = len(self.characters_plan.get('new', []))
            if new_count > 0:
                logger.info(f"[细纲会话] 规划新增{new_count}个角色")
                
        except Exception as e:
            logger.error(f"[细纲会话] 角色规划失败: {e}")
            self.characters_plan = self._get_default_character_plan()
    
    def _build_character_dialog_prompt(self) -> str:
        """构建角色规划对话提示词"""
        # 格式化已有角色
        existing_text = "\n".join([
            f"- {c.get('name')}: {c.get('role', '未知')} - {', '.join(c.get('traits', [])[:3])}"
            for c in self.existing_characters[:20]
        ])
        
        # 格式化情节大纲
        chapters = self.chapters_plan.get('chapters', [])
        chapters_text = "\n".join([
            f"第{ch['chapter_number']}章: {ch.get('event', '')[:40]}..."
            for ch in chapters[:15]
        ])
        
        return f"""
## 任务：为30章战术规划角色出场

### 已有角色列表（优先使用）
{existing_text}

### 本批次情节大纲
{chapters_text}

### 角色规划要求
1. **核心角色**（每章必出场）：主角、标配角色
2. **重要配角**（关键章节出场）：优先从已有角色选择
3. **次要配角**（临时出场）：可新增，但需有后续安排
4. **新角色**（谨慎新增）：仅当已有角色不足时新增，每批最多2个

### 输出格式（JSON）
```json
{{
    "core": [{{"name": "...", "appearance": "每章必出场"}}],
    "major": [{{"name": "...", "first_chapter": 5, "key_chapters": [5,10]}}],
    "minor": [{{"name": "...", "first_chapter": 3}}],
    "new": [{{"name": "...", "first_chapter": 25, "purpose": "...", "reason": "..."}}]
}}
```
"""
    
    def _get_character_system_prompt(self) -> str:
        return """你是专业的角色设计师。
基于已有角色列表和情节大纲，规划每章的角色出场。
原则：优先复用已有角色，避免频繁引入新角色。
输出严格JSON格式。"""
    
    def _parse_character_response(self, response) -> Dict:
        """解析AI返回的角色规划"""
        try:
            if isinstance(response, dict):
                return response
            
            text = str(response)
            json_match = re.search(r'\{[\s\S]*\}', text)
            if json_match:
                return json.loads(json_match.group())
        except Exception as e:
            logger.error(f"解析角色规划失败: {e}")
        
        return self._get_default_character_plan()
    
    def _get_default_character_plan(self) -> Dict:
        """默认角色规划"""
        protagonist = self._get_protagonist_name()
        return {
            'core': [{'name': protagonist, 'appearance': '每章必出场'}],
            'major': [],
            'minor': [],
            'new': []
        }
    
    def _merge_blueprint(self):
        """合并情节和角色规划"""
        chapters = self.chapters_plan.get('chapters', [])
        
        # 为每章添加角色出场信息
        for chapter in chapters:
            ch_num = chapter.get('chapter_number')
            chapter['assigned_characters'] = self._get_characters_for_chapter(ch_num)
        
        self.final_blueprint = {
            'chapters': chapters,
            'characters': self.characters_plan,
            'session_id': self.session_id,
            'start_chapter': self.start_chapter,
            'end_chapter': self.end_chapter,
            'generated_at': datetime.now().isoformat(),
            'rounds_completed': self.current_round
        }
    
    def _get_characters_for_chapter(self, chapter_num: int) -> Dict:
        """获取某章应出场的角色"""
        assigned = {'core': [], 'major': [], 'minor': []}
        
        # 核心角色
        for char in self.characters_plan.get('core', []):
            assigned['core'].append(char.get('name'))
        
        # 重要配角
        for char in self.characters_plan.get('major', []):
            first_ch = char.get('first_chapter', self.start_chapter)
            key_chapters = char.get('key_chapters', [])
            
            if chapter_num >= first_ch:
                if not key_chapters or chapter_num in key_chapters:
                    assigned['major'].append(char.get('name'))
        
        # 次要配角
        for char in self.characters_plan.get('minor', []):
            if chapter_num == char.get('first_chapter'):
                assigned['minor'].append(char.get('name'))
        
        return assigned
    
    def _get_protagonist_name(self) -> str:
        user_choices = self.novel_data.get('user_choices', {})
        final_plan = self.novel_data.get('final_plan', {})
        return user_choices.get('protagonist_name', 
               final_plan.get('protagonist_name', '主角'))
    
    def _get_stage_goal(self) -> Dict:
        return self.novel_data.get('stage_goals', [{}])[0]
    
    def _get_previous_summary(self) -> Dict:
        return self.novel_data.get('batch_summary', {})


# 全局会话管理
_tactical_sessions: Dict[str, TacticalDialogSession] = {}

def create_tactical_dialog_session(*args, **kwargs) -> TacticalDialogSession:
    session = TacticalDialogSession(*args, **kwargs)
    _tactical_sessions[session.session_id] = session
    return session

def get_tactical_dialog_session(session_id: str) -> Optional[TacticalDialogSession]:
    return _tactical_sessions.get(session_id)
```

def _build_full_prompt(self, chapter_num: int, chapter_plan: Dict, ...):
    """构建章节生成提示词，加入角色约束"""
    
    # 获取本章角色规划
    assigned_chars = chapter_plan.get('assigned_characters', {})
    
    # 构建角色约束提示词
    char_constraint = ""
    if assigned_chars:
        core = assigned_chars.get('core', [])
        major = assigned_chars.get('major', [])
        minor = assigned_chars.get('minor', [])
        
        char_constraint = f"""
### 本章角色出场规划（必须严格遵守）
🔴 核心角色（必须出场）: {', '.join(core)}
🟡 重要配角（建议出场）: {', '.join(major) if major else '无'}
🟢 次要配角（可选出场）: {', '.join(minor) if minor else '无'}

⚠️ 禁止：创造上述列表之外的有名新角色（路人/NPC可用通称）
"""
    
    # 构建完整提示词
    full_prompt = char_constraint + chapter_prompt
    
    return full_prompt
```
